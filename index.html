      if (!raw) return defaultValue;

      const data = JSON.parse(raw);

      return key === 'reviews' ? data.map(r => ({ ...r, is_pinned: r.is_pinned || false })) : data;

    }

    function setLocalData(key, value) { localStorage.setItem(`tarot_${key}`, JSON.stringify(value)); }


    if (USE_LOCAL_STORAGE) {

      if (!localStorage.getItem('tarot_invite_codes')) setLocalData('invite_codes', [{ id: 1, code: 'TARO', is_used: false, created_at: new Date().toISOString() }]);

      if (!localStorage.getItem('tarot_reviews')) setLocalData('reviews', []);

    }


    async function supabaseFetch(path, options = {}) {

      if (USE_LOCAL_STORAGE) throw new Error('本地模式');

      const url = `${SUPABASE_URL}/rest/v1/${path}`;

      const headers = { 'apikey': SUPABASE_ANON_KEY, 'Authorization': `Bearer ${SUPABASE_ANON_KEY}`, 'Content-Type': 'application/json', 'Prefer': 'return=representation', ...options.headers };

      const res = await fetch(url, { ...options, headers });

      if (!res.ok) throw new Error(`请求失败: ${res.status}`);

      const text = await res.text();

      return text ? JSON.parse(text) : null;

    }


    async function supabaseRPC(functionName, body) {

      if (USE_LOCAL_STORAGE) return simulateSubmitReview(body);

      const url = `${SUPABASE_URL}/rest/v1/rpc/${functionName}`;

      const headers = { 'apikey': SUPABASE_ANON_KEY, 'Authorization': `Bearer ${SUPABASE_ANON_KEY}`, 'Content-Type': 'application/json' };

      const res = await fetch(url, { method: 'POST', headers, body: JSON.stringify(body) });

      if (!res.ok) throw new Error(`RPC失败: ${res.status}`);

      const text = await res.text();

      return text ? JSON.parse(text) : null;

    }


    function simulateSubmitReview({ p_invite_code, p_nickname, p_rating, p_content }) {

      const codes = getLocalData('invite_codes', []);

      const codeIndex = codes.findIndex(c => c.code === p_invite_code && !c.is_used);

      if (codeIndex === -1) return { success: false, message: '邀请码无效或已被使用' };

      codes[codeIndex].is_used = true;

      setLocalData('invite_codes', codes);

      const reviews = getLocalData('reviews', []);

      reviews.unshift({ id: Date.now(), nickname: p_nickname || '匿名', rating: p_rating, content: p_content, is_hidden: false, is_pinned: false, created_at: new Date().toISOString() });

      setLocalData('reviews', reviews);

      return { success: true, message: '评价提交成功' };

    }

    function switchTab(tabName) {

      document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));

      document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));

      document.getElementById(`tab-${tabName}`).classList.add('active');

      event.target.classList.add('active');

      if (tabName === 'reviews') loadReviews();

    }


    function openModal(id) { document.getElementById(id).style.display = 'flex'; document.body.style.overflow = 'hidden'; }

    function closeModal(id) { document.getElementById(id).style.display = 'none'; document.body.style.overflow = ''; }

    function openReviewModal() {

      document.getElementById('reviewContent').value = '';

      document.getElementById('reviewNickname').value = '';

      document.getElementById('inviteCodeInput').value = '';

      document.getElementById('submitBtn').disabled = false;

    }

    function closeReviewModal() { closeModal('reviewModal'); }

    function openAdminPanel() { closeModal('adminLoginModal'); openModal('adminPanelModal'); loadAdminData(); }

    function showToast(msg, type = 'info') {

      const toast = document.createElement('div');

      toast.className = `toast ${type === 'success' ? 'toast-success' : type === 'error' ? 'toast-error' : ''}`;

      toast.textContent = msg;

      document.body.appendChild(toast);

      setTimeout(() => { toast.style.opacity = '0'; toast.style.transition = '0.3s'; setTimeout(() => toast.remove(), 300); }, 2500);

    }


    async function loadReviews() {

      const container = document.getElementById('reviewsList');

      container.innerHTML = '<div style="text-align:center;padding:28px;"><div class="loading-spinner"></div></div>';

      try {

        let reviews;

        if (USE_LOCAL_STORAGE) {

          reviews = getLocalData('reviews', []).filter(r => !r.is_hidden).sort((a,b) => { if (a.is_pinned && !b.is_pinned) return -1; if (!a.is_pinned && b.is_pinned) return 1; return new Date(b.created_at) - new Date(a.created_at); });

        } else {

          reviews = await supabaseFetch('reviews?select=*&is_hidden=eq.false&order=is_pinned.desc,created_at.desc&limit=50');

        }

        renderReviews(reviews || []);

        updateVisitorCount((reviews || []).length);

      } catch (e) {

        container.innerHTML = '<div class="empty-reviews">暂时无法加载评价，请稍后再试</div>';

        document.getElementById('visitorFeedbackText').textContent = '';

      }

    }


    function renderReviews(reviews) {

      const container = document.getElementById('reviewsList');

      if (!reviews.length) { container.innerHTML = '<div class="empty-reviews">暂无反馈，期待你的第一条真实感受</div>'; return; }

      container.innerHTML = reviews.map(r => {

        const initial = (r.nickname || '匿')[0].toUpperCase();

        const time = formatTime(r.created_at);

        return `<div class="review-card ${r.is_pinned ? 'pinned' : ''}">

          ${r.is_pinned ? '<span class="pinned-badge">置顶</span>' : ''}

          <div class="review-header"><div class="review-avatar">${initial}</div><span class="review-nickname">${escapeHTML(r.nickname || '匿名')}</span><span class="review-time">${time}</span></div>

          ${r.content ? `<div class="review-content">${escapeHTML(r.content)}</div>` : ''}

        </div>`;

      }).join('');

    }


    function updateVisitorCount(count) { document.getElementById('visitorFeedbackText').textContent = count > 0 ? `已有${count}位来访者，在这里留下他们的反馈。` : '期待第一位来访者的反馈'; }

    function formatTime(d) { if (!d) return ''; const n = new Date(), diff = n - new Date(d); const m = Math.floor(diff/60000), h = Math.floor(diff/3600000), day = Math.floor(diff/86400000); if (m<1) return '刚刚'; if (m<60) return m+'分钟前'; if (h<24) return h+'小时前'; if (day<30) return day+'天前'; return `${new Date(d).getFullYear()}-${String(new Date(d).getMonth()+1).padStart(2,'0')}-${String(new Date(d).getDate()).padStart(2,'0')}`; }

    function escapeHTML(s) { const div = document.createElement('div'); div.textContent = s; return div.innerHTML; }


    async function submitReview() {

      const code = document.getElementById('inviteCodeInput').value.trim();

      const nickname = document.getElementById('reviewNickname').value.trim() || '匿名';

      const content = document.getElementById('reviewContent').value.trim();

      const btn = document.getElementById('submitBtn');

      if (!code) return showToast('请输入邀请码', 'error');

      if (!content) return showToast('请填写评价内容', 'error');

      btn.disabled = true; btn.textContent = '提交中…';

      try {

        const result = await supabaseRPC('submit_review', { p_invite_code: code, p_nickname: nickname, p_rating: 5, p_content: content });

        if (result?.success) { showToast('反馈提交成功！', 'success'); closeReviewModal(); loadReviews(); }

        else showToast(result?.message || '邀请码无效或已使用', 'error');

      } catch (e) { showToast('提交失败，请重试', 'error'); }

      finally { btn.disabled = false; btn.textContent = '提交反馈'; }

    }


    function adminLogin() {

      if (document.getElementById('adminPassword').value === ADMIN_PASSWORD) { isAdmin = true; showToast('已进入管理后台', 'success'); openAdminPanel(); }

      else showToast('密码错误', 'error');

    }


    async function loadAdminData() {

      try {

        let codes, reviews;

        if (USE_LOCAL_STORAGE) { codes = getLocalData('invite_codes', []).sort((a,b)=> new Date(b.created_at)-new Date(a.created_at)); reviews = getLocalData('reviews', []).sort((a,b)=> new Date(b.created_at)-new Date(a.created_at)); }

        else { codes = await supabaseFetch('invite_codes?select=*&order=created_at.desc&limit=100'); reviews = await supabaseFetch('reviews?select=*&order=created_at.desc&limit=100'); }

        renderAllCodes(codes || []); renderAdminReviews(reviews || []);

      } catch (e) { showToast('加载管理数据失败', 'error'); }

    }


    function renderAllCodes(codes) { document.getElementById('allCodesList').innerHTML = codes.length ? codes.map(c => `<span class="code-tag ${c.is_used?'used':''}" onclick="copyCode('${c.code}')">${c.code}${c.is_used?' (已用)':''}</span>`).join('') : '暂无邀请码'; }

    function renderAdminReviews(reviews) {

      const container = document.getElementById('adminReviewsList');

      if (!reviews.length) { container.innerHTML = '<div style="padding:8px;color:#999;">暂无评价</div>'; return; }

      container.innerHTML = reviews.map(r => `

        <div class="admin-review-row">

          <div class="admin-review-info"><b>${escapeHTML(r.nickname||'匿名')}</b>${r.is_pinned?'<span class="pin-indicator">置顶</span>':''}<span style="color:#999;font-size:0.6875rem;margin-left:6px;">${formatTime(r.created_at)}</span>${r.is_hidden?'<span style="color:#e65050;font-size:0.625rem;">[已隐藏]</span>':''}<div style="font-size:0.75rem;color:#555;">${escapeHTML((r.content||'').substring(0,30))}…</div></div>

          <div class="admin-review-actions">

            <button class="btn btn-sm ${r.is_pinned?'btn-pin':'btn-outline'}" onclick="togglePin(${r.id},${r.is_pinned})">${r.is_pinned?'取消置顶':'置顶'}</button>

            <button class="btn btn-sm ${r.is_hidden?'btn-outline':'btn-danger'}" onclick="toggleHidden(${r.id},${r.is_hidden})">${r.is_hidden?'显示':'隐藏'}</button>

            <button class="btn btn-sm btn-warning" onclick="deleteReview(${r.id})">删除</button>

          </div>

        </div>`).join('');

    }


    async function togglePin(id, current) {

      if (USE_LOCAL_STORAGE) { const reviews = getLocalData('reviews', []); const idx = reviews.findIndex(r => r.id === id); if (idx !== -1) reviews[idx].is_pinned = !current; setLocalData('reviews', reviews); }

      else await supabaseFetch(`reviews?id=eq.${id}`, { method: 'PATCH', body: JSON.stringify({ is_pinned: !current }) });

      loadAdminData(); loadReviews(); showToast(current ? '已取消置顶' : '已置顶', 'success');

    }

    async function toggleHidden(id, current) {

      if (USE_LOCAL_STORAGE) { const reviews = getLocalData('reviews', []); const idx = reviews.findIndex(r => r.id === id); if (idx !== -1) reviews[idx].is_hidden = !current; setLocalData('reviews', reviews); }

      else await supabaseFetch(`reviews?id=eq.${id}`, { method: 'PATCH', body: JSON.stringify({ is_hidden: !current }) });

      loadAdminData(); loadReviews(); showToast(current ? '评价已显示' : '评价已隐藏', 'success');

    }

    async function deleteReview(id) {

      if (!confirm('确定要永久删除这条评价吗？')) return;

      if (USE_LOCAL_STORAGE) { let reviews = getLocalData('reviews', []); reviews = reviews.filter(r => r.id !== id); setLocalData('reviews', reviews); }

      else await supabaseFetch(`reviews?id=eq.${id}`, { method: 'DELETE' });

      showToast('评价已删除', 'success'); loadAdminData(); loadReviews();

    }

    async function generateInviteCodes() {

      const count = Math.min(20, Math.max(1, parseInt(document.getElementById('genCodeCount').value) || 5));

      const existingCodes = USE_LOCAL_STORAGE ? getLocalData('invite_codes', []).map(c => c.code) : [];

      const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';

      const newCodes = [];

      while (newCodes.length < count) {

        let code = ''; for (let i=0;i<4;i++) code += chars[Math.floor(Math.random()*chars.length)];

        if (!existingCodes.includes(code) && !newCodes.find(c => c.code === code)) newCodes.push({ code, is_used: false });

        if (newCodes.length + existingCodes.length >= 20000) break;

      }

      if (!newCodes.length) { showToast('无法生成更多邀请码', 'error'); return; }

      try {

        if (USE_LOCAL_STORAGE) {

          const existing = getLocalData('invite_codes', []), timestamp = Date.now();

          const codesToAdd = newCodes.map((c,i) => ({ id: timestamp+i, ...c, created_at: new Date().toISOString() }));

          setLocalData('invite_codes', [...existing, ...codesToAdd]);

          document.getElementById('newCodesList').innerHTML = codesToAdd.map(c => `<span class="code-tag" onclick="copyCode('${c.code}')">${c.code}</span>`).join('');

        } else {

          const result = await supabaseFetch('invite_codes', { method: 'POST', body: JSON.stringify(newCodes) });

          const createdCodes = result || newCodes;

          document.getElementById('newCodesList').innerHTML = createdCodes.map(c => `<span class="code-tag" onclick="copyCode('${c.code}')">${c.code}</span>`).join('');

        }

        showToast(`已生成 ${newCodes.length} 个邀请码`, 'success'); loadAdminData();

      } catch (e) { showToast('生成失败', 'error'); }

    }

    function copyCode(code) {

      navigator.clipboard?.writeText(code).then(() => showToast('已复制: ' + code)).catch(() => {

        const ta = document.createElement('textarea'); ta.value = code; document.body.appendChild(ta); ta.select();

        document.execCommand('copy'); document.body.removeChild(ta); showToast('已复制');

      });

    }
