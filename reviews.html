    function formatTime(d) {

      if (!d) return '';

      const n = new Date(), diff = n - new Date(d);

      const m = Math.floor(diff/60000), h = Math.floor(diff/3600000), day = Math.floor(diff/86400000);

      if (m<1) return '刚刚';

      if (m<60) return m+'分钟前';

      if (h<24) return h+'小时前';

      if (day<30) return day+'天前';

      return `${new Date(d).getFullYear()}-${String(new Date(d).getMonth()+1).padStart(2,'0')}-${String(new Date(d).getDate()).padStart(2,'0')}`;

    }


    function escapeHTML(s) {

    }

    async function generateInviteCodes() {

      const count = Math.min(20, Math.max(1, parseInt(document.getElementById('genCodeCount').value) || 5));

      const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';

      const newCodes = [];

        newCodes.push({ code, is_used: false });

      }

      try {

        const result = await supabaseFetch('invite_codes', { method: 'POST', body: JSON.stringify(newCodes) });

    function copyCode(code) {

      navigator.clipboard?.writeText(code).then(() => showToast('已复制: ' + code)).catch(() => {

        const ta = document.createElement('textarea'); ta.value = code; document.body.appendChild(ta); ta.select();

        document.execCommand('copy'); document.body.removeChild(ta); showToast('已复制');

      });

    }
